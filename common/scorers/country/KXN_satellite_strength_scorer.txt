KXN_Satellite_Strength_Scorer = {
	targets = {
		targets_dynamic = yes
		target_non_existing = no
		target_array = KXN.subjects
		target_trigger = {
			ROOT = {
				tag = KXN
				is_subject = no
				has_government = fascism
				NOT = { has_global_flag = KXN_Rutilant_Revolt_Happened }
			}
			FROM = {
				original_tag = KXN
				KXN_IS_SATELLITE_TRIGGER = yes
				NOT = { has_country_flag = KXN_Colony_Flag }
			}
		}
		score = {
			base = 1
			modifier = {
				always = yes 
				add = has_added_tension_amount
			}
			modifier = {
				always = yes 
				add = num_ships
			}
			modifier = {
				always = yes 
				add = num_battalions
			}
			modifier = {
				always = yes 
				add = num_deployed_planes
			}
			modifier = {
				always = yes 
				add = num_of_factories
			}
			modifier = {
				always = yes 
				add = num_of_nukes
			}
		}
	}
}

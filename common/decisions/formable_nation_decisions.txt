form_quaggatai_kingdom_category = {

	form_quaggatai_kingdom = {
		icon = generic_form_nation
		
		fire_only_once = yes
		
		available = {
			is_subject = no
			CGI = {
				all_core_state = {
					owner = {
						tag = ROOT
					}
				}
			}
			BKH = {
				all_core_state = {
					owner = {
						tag = ROOT
					}
				}
			}
			CTH = {
				all_core_state = {
					owner = {
						tag = ROOT
					}
				}
			}
			CVA = {
				all_core_state = {
					OR = {
						state = 1092
						owner = {
							tag = ROOT
						}
					}
				}
			}
			CDR = {
				all_core_state = {
					owner = {
						tag = ROOT
					}
				}
			}
		}

		visible = {
			is_subject = no
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision complete form_quaggatai_kingdom"
			
			set_cosmetic_tag = CGI_KINGDOM
			
			if = {
				limit = {
					CGI = {
						any_core_state = {
							NOT = { is_core_of = ROOT }
						}
					}
				}
				CGI = { add_to_ROOT_potential_original_cores = yes } # Quaggatai
			}
			if = {
				limit = {
					BKH = {
						any_core_state = {
							NOT = { is_core_of = ROOT }
						}
					}
				}
				BKH = { add_to_ROOT_potential_original_cores = yes } # Quaggatai
			}
			if = {
				limit = {
					CTH = {
						any_core_state = {
							NOT = { is_core_of = ROOT }
						}
					}
				}
				CTH = { add_to_ROOT_potential_original_cores = yes } # Quaggatai
			}
			if = { # Special case in case CTH doesn't have cores anymore
				limit = {
					875 = { # Colthage state
						NOT = { is_core_of = CTH } # CTH cores have been removed
					}
				}
				random_country_with_original_tag = {
					original_tag_to_check = CTH
					limit = {
						875 = {
							is_core_of = PREV
						}
					}
					add_to_ROOT_potential_cores = yes
				}
			}
			if = {
				limit = {
					CVA = {
						any_core_state = {
							NOT = { is_core_of = ROOT }
						}
					}
				}
				CVA = { add_to_ROOT_potential_original_cores = yes } # Quaggatai
			}
			if = {
				limit = {
					CDR = {
						any_core_state = {
							NOT = { is_core_of = ROOT }
						}
					}
				}
				CDR = { add_to_ROOT_potential_original_cores = yes } # Quaggatai
			}
		}

		ai_will_do = {
			factor = 200
		}
	}
}

form_the_socialist_world_republic = {
	form_the_socialist_world_republic_decision = {
		icon = red_star
		cost = 25
		ai_will_do = {
			factor = 100
		}
		fire_only_once = yes
		visible = {
			has_government = communism
			check_variable = { num_owned_controlled_states = global.wc_states_required }
		}
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision form_the_socialist_world_republic_decision"
			drop_cosmetic_tag = yes
			if = {
				limit = {
					OR = {
						has_country_leader_ideology = socialist_monarchism
						tag = BRZ
					}
				}
				set_cosmetic_tag = COE_world
			}
			else = {
				set_cosmetic_tag = COM_world
			}
		}
	}
}

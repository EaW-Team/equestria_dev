			}	
		}
		
		effects = {
			
			button_right_major_list_click = {
				add_to_variable = { EQS_coc_current_majors_min = 1 }
				clamp_variable = { var = EQS_coc_current_majors_min min = 0 max = EQS_coc_majors_list^num }
				EQS_show_majors_next = yes 
			}
			button_right_major_list_shift_click = {
				add_to_variable = { EQS_coc_current_majors_min = 5 }
				clamp_variable = { var = EQS_coc_current_majors_min min = 0 max = EQS_coc_majors_list^num }
				EQS_show_majors_next = yes 
			}
			button_left_major_list_click = {
				add_to_variable = { EQS_coc_current_majors_min = -1 }
				clamp_variable = { var = EQS_coc_current_majors_min min = 0 max = EQS_coc_majors_list^num }
				EQS_show_majors_next = yes 
			}
			button_left_major_list_shift_click = {
				add_to_variable = { EQS_coc_current_majors_min = -5 }
				clamp_variable = { var = EQS_coc_current_majors_min min = 0 max = EQS_coc_majors_list^num }
				EQS_show_majors_next = yes 
			}
		}
		
		triggers = {
			button_right_major_list_click_enabled = {				
				check_expr = { 	
					value = EQS_coc_current_majors_min
					add = 1
					less_than = EQS_coc_majors_list^num
				}
			}
			button_left_major_list_click_enabled = {
				check_variable = { EQS_coc_current_majors_min > 0 }
			}

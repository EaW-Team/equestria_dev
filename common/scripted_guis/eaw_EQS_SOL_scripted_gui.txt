scripted_gui = {
    SOL_great_works_toggle_container = {
        parent_window_token = politics_tab
        window_name = "SOL_great_works_toggle"
        context_type = player_context

        visible = {
            tag = EQS
            is_EQS_SOL = yes
        }

        effects = {
            SOL_great_works_toggle_button_click = {
                if = {
                    limit = {
                        has_country_flag = SOL_great_works_window_open
                    }
                    clr_country_flag = SOL_great_works_window_open
                }
                else = {
                    set_country_flag = SOL_great_works_window_open
                }
            }
        }
    }
    SOL_great_works_window_container = {
        context_type = player_context
        window_name = "SOL_great_works_window"

        visible = {
            tag = EQS
            is_EQS_SOL = yes
            has_country_flag = SOL_great_works_window_open
        }

        effects = {
            SOL_great_works_window_close_click = {
                clr_country_flag = SOL_great_works_window_open
            }
        }
    }
}

scripted_gui = {
	solitaire_GUI = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "solitaire_GUI_window"
		dynamic_lists = {
			solitaire_GUI_grid = {
				array = ROOT.cards_GUI_array
				entry_container = solitaire_GUI_entry
				value = point_v
			}
		}
		triggers = {
			solitaire_GUI_card_button_visible = {
				set_temp_variable = {
					temp_y = {
						value = y_pos^point_v
						add = ROOT.cards_GUI_y_pos^1
					}
				}
				any_of = {
					array = ROOT.cards_GUI_array
					value = v_temp
					check_variable = { ROOT.x_pos^v_temp = x_pos^point_v }
					NOT = {
						is_v_compatible_with_v_temp = yes
						AND = {
							check_variable = { ROOT.x_pos^v_temp = x_pos^point_v }
							check_variable = { ROOT.y_pos^v_temp = temp_y }
						}
					}
				}
			}
		}
		effects = {
			solitaire_GUI_card_button_click = {
				if = {
					limit = {
						NOT = { has_variable = selected_card_value }
					}
					set_variable = { selected_card_value = point_v }
				}
				else_if = {
					limit = {
						NOT = { check_variable = { x_pos^point_v = ROOT.cards_GUI_x_pos^7 } }
					}
					set_temp_variable = { 
						temp_index = {
							value = x_pos^point_v
							divide = ROOT.cards_GUI_x_pos^1
						}
					}
					set_variable = { ROOT.x_pos^selected_card_value = ROOT.cards_GUI_x_pos^temp_index }
					set_temp_variable = { 
						temp_index = {
							value = y_pos^point_v
							divide = ROOT.cards_GUI_y_pos^1
							add = 1
						}
					}
					set_variable = { ROOT.y_pos^selected_card_value = ROOT.cards_GUI_y_pos^temp_index }
					clear_variable = selected_card_value
				}
			}
		}
		properties = {
			solitaire_GUI_card_icon = {
				frame = point_v
				x = ROOT.x_pos^point_v
				y = ROOT.y_pos^point_v
			}
			solitaire_GUI_card_button = {
				frame = point_v
				x = ROOT.x_pos^point_v
				y = ROOT.y_pos^point_v
			}
		}
	}
}

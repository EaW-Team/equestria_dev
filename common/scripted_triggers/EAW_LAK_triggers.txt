LAK_BRD_current_and_potential_enemies = {
	is_on_same_continent_as = ROOT
	OR = {
		exists_outside_sphere_of_influence = yes
		has_war_with = ROOT
	}
	NOT = { tag = DED }
	OR = {
		has_wargoal_against = ROOT
		if = {
			limit = {
				has_completed_focus = LAK_BRD_no_compromise
			}
			has_same_ideology = no
			has_offensive_war = yes
		}
		else = {
			always = no
		}
		any_controlled_state = { is_claimed_by = ROOT }
		ROOT = { any_controlled_state = { is_claimed_by = PREV.PREV } }
	}
}

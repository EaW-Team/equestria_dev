is_v_compatible_with_v_temp = {
	if = {
		limit = {
			OR = {
				check_variable = { v < 13 }
				AND = {
					check_variable = { v_temp > 24 }
					check_variable = { v_temp < 37 }
				}
			}
		}
		OR = {
			check_variable = { v > 12 }
			AND = {
				check_variable = { v_temp < 25 }
				check_variable = { v_temp > 36 }
			}
		}
	}
	else = {
		OR = {
			check_variable = { v < 13 }
			AND = {
				check_variable = { v_temp > 24 }
				check_variable = { v_temp < 37 }
			}
		}
	}
}


scripted_diplomatic_actions = {

	recall_volunteers = {
	
		allowed = {
			ROOT = { is_ai = no }
		}
		visible = {
			FROM = {
				has_volunteers_amount_from = { tag = ROOT count > 0 }
			}
		}
		selectable = {
			# doing this for nicer tooltips
			# commented because it is made invisible if they dont have volunteers from us
			#if = {
			#	limit = {
			#		NOT = { has_volunteers_amount_from = { tag = ROOT count > 0 } }
			#	}
			#	custom_trigger_tooltip = {
			#		tooltip = RECALL_VOLUNTEERS_TOOLTIP
			#		always = no
			#	}
			#}
			if = {
                limit = {
                    OR = {
                        AND = {
                            ROOT = {
                                has_relation_modifier = {
                                    target = EQS
                                    modifier = JUN_mercs_for_peace
                                }
                            }
                            FROM = {
                                original_tag = EQS
                                has_government = democratic
                            }
                        }
                    }
                }
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP_DISABLED
					always = no
				}
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							ROOT = {
								has_idea = GRF_send_mercs
							}
							OR = {
								ROOT = { has_country_flag = agreed_to_mercs_@FROM }
								FROM = {
									OR = {
										has_idea = GRF_got_mercs
										has_idea = GRF_got_mercs_delayed
									}
								}
							}
						}
						AND = {
							ROOT = {
								has_idea = TRD_mercs_idea
							}
							OR = {
								ROOT = { has_country_flag = agreed_to_mercs_@FROM }
								FROM = {
									OR = {
										has_idea = TRD_got_mercs
										has_idea = TRD_got_mercs_delayed
									}
								}
							}
						}
					}
					ROOT = { has_country_flag = mercs_one_year_@FROM }
				}
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP_G
					always = no
				}
			}
			else = {
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP_G
					always = yes
				}
			}
		}

		cost = 0
		command_power = 0

		show_acceptance_on_action_button = no

		icon = 1

		complete_effect = {
			ROOT = { recall_volunteers_from = PREV }
		}

		ai_acceptance = {
			condition = {
				base = 100
			}
		}
	}
}

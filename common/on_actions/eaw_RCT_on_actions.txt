on_actions = {
	on_declare_war = {
		effect = {
			if = {#If someone goes to war with Nova Whirl while they're infiltrated with her partisans
				limit = {
					FROM = {
						original_tag = RCT
					}
					ROOT = {
						OR = {
							has_idea = RCT_militants1
							has_idea = RCT_militants2
							has_idea = RCT_militants3
						}
					}
				}
				ROOT = {
					country_event = { id = rivercity.1920 days = 2}
				}
			}
			else_if = {#If Nova Whirl goes to war with someone while they're infiltrated with her partisans
				limit = {
					FROM = {
						OR = {
							has_idea = RCT_militants1
							has_idea = RCT_militants2
							has_idea = RCT_militants3
						}
					}
					ROOT = {
						original_tag = RCT
					}
				}
				FROM = {
					country_event = { id = rivercity.1920 days = 2}
				}
			}
		}
	}

	on_puppet = {
		effect = {
			if = {#If RCT has done Friendship Without Borders, this will apply it to any future vassals they make
				limit = {
					FROM = {
						original_tag = RCT
						has_completed_focus = RCT_friendship_without_borders
					}
					ROOT = {
						NOT = {
							has_idea = RCT_friendship_without_borders_idea
						}
					}
				}
				add_ideas = RCT_friendship_without_borders_idea
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			if = {#If RCT has done Friendship Without Borders, this will apply it to any future vassals they make
				limit = {
					FROM = {
						original_tag = RCT
						has_completed_focus = RCT_friendship_without_borders
					}
					ROOT = {
						NOT = { has_idea = RCT_friendship_without_borders_idea }
					}
				}
				add_ideas = RCT_friendship_without_borders_idea
			}
		}
	}

	on_civil_war_end = {
		effect = {
			if = {
				limit = { original_tag = RCT }
				set_capital = { state = 619 remember_old_capital = no }
			}
		}
	}

	on_daily_RCT = {
		effect = {
			IF = {
				limit = {
					has_political_power < 1
					has_idea = RCT_election_campaign_pp_cost_hidden_idea
				}
				remove_ideas = RCT_election_campaign_pp_cost_hidden_idea
			}
			# Democratic Election Campaigning
			IF = { # HSR
				limit = {
					check_variable = {
						var = RCT_campaigning_for_party
						value = 1
						compare = equals
					}
				}
				add_to_variable = {
					var = RCT_hsr_votes_var
					value = 0.05 #Edit - RCT_hsr_election_campaign_tt /AND/ RCT_skr_election_campaign_tt /AND/ RCT_hsr_votes_change_var /AND/ RCT_skr_votes_change_var - if changed
				}
				add_to_variable = {
					var = RCT_skr_votes_var
					value = -0.05 #Edit - RCT_hsr_election_campaign_tt /AND/ RCT_skr_election_campaign_tt /AND/ RCT_hsr_votes_change_var /AND/ RCT_skr_votes_change_var - if changed
				}
			}
			IF = { # SKR
				limit = {
					check_variable = {
						var = RCT_campaigning_for_party
						value = 2
						compare = equals
					}
				}
				add_to_variable = {
					var = RCT_skr_votes_var
					value = 0.05 #Edit - RCT_hsr_election_campaign_tt /AND/ RCT_skr_election_campaign_tt /AND/ RCT_hsr_votes_change_var, RCT_skr_votes_change_var in RCT_hsr_campaign - if changed
				}
				add_to_variable = {
					var = RCT_hsr_votes_var
					value = -0.05 #Edit - RCT_hsr_election_campaign_tt /AND/ RCT_skr_election_campaign_tt /AND/ RCT_hsr_votes_change_var, RCT_skr_votes_change_var in RCT_hsr_campaign - if changed
				}
			}
			IF = { # failsafe for campaigning not ending
				limit = {
					OR = {
						NOT = {
							OR = {
								check_variable = {
									var = RCT_campaigning_for_party
									value = 1
									compare = equals
								}
								check_variable = {
									var = RCT_campaigning_for_party
									value = 2
									compare = equals
								}
							}
						}
						NOT = {
							has_government = communism
							has_government = democratic
						}
					}
				}
				remove_ideas = RCT_election_campaign_pp_cost_hidden_idea
			}
			clamp_variable = {
				var = RCT_hsr_votes_var
				min = 0
				max = 100
			}
			clamp_variable = {
				var = RCT_skr_votes_var
				min = 0
				max = 100
			}
		}
	}
}

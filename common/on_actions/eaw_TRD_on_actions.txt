on_actions = {
	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = { original_tag = TRD }
				}
				every_other_country = {
					limit = {
						OR = {
							has_idea = TRD_trading_with_skyfall_1
							has_idea = TRD_trading_with_skyfall_2
						}
					}
					add_opinion_modifier = {
						target = ROOT
						modifier = TRD_disrupted_trade
					}
				}
			}
			if = {
				limit = {
					TRD = { has_idea = TRD_mercs_idea }
				}
				log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] TRD on_annex GRF_recalc_mercs"
				TRD = { country_event = { id = utility.22 days = 1 } }
			}
		}
	}

	on_peace = {
		effect = {
			if = {
				limit = {
					TRD = { has_idea = TRD_mercs_idea }
					has_volunteers_amount_from = {
						tag = TRD
						count > 0
					}
				}
				log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] TRD on_peace GRF_recalc_mercs"
				TRD = { country_event = { id = utility.22 days = 1 } }
			}
		}
	}

	on_monthly_TRD = {
		effect = {
			if = {
				limit = {
					has_idea = TRD_mercs_idea
				}
				GRF_recalc_mercs = yes
			}
		}
	}

	on_send_volunteers = {
		effect = {
			if = {
				limit = {
					has_idea = TRD_mercs_idea
				}
				log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] TRD on_send_volunteers GRF_recalc_mercs"
				country_event = { id = utility.22 days = 1 }
			}
		}
	}

	on_recall_volunteers = {
		effect = {
			if = {
				limit = {
					has_idea = TRD_mercs_idea
				}
				log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] TRD on_recall_volunteers GRF_recalc_mercs"
				country_event = { id = utility.22 days = 1 }
			}
		}
	}
}

@BLT_SKYSTRIKER_BONUS_CAP = 35

on_actions = {
	#When Harira's invitation arrives
	on_startup = {
		effect = {
			BLT = {
				country_lock_all_division_template = yes
				random_list = {
					60 = {
						country_event = { id = boltria.2 days = 10 }
					}
					25 = {
						country_event = { id = boltria.2 days = 300 }
					}
					15 = {
						country_event = { id = boltria.2 days = 800 }
					}
				}
			}
		}
	}

	#BLT and CME immediately annex each other to avoid peace conference shenanigans
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					FROM = { original_tag = BLT }
					ROOT = { original_tag = CME }
				}
				BLT = {
					annex_country = { target = ROOT }
				}
			}
		}
	}

	on_declare_war = {
		effect = {
			#Gallant trait effect
			if = {
				limit = {
					FROM = { original_tag = BLT }
					BLT = {
						BLT_boltrus_is_gallant = yes
						NOT = { has_idea = BLT_rally_around_the_king }
					}
				}
				BLT = { add_ideas = BLT_rally_around_the_king }
			}
			
			#Harira's advisor ability
			if = {
				limit = {
					original_tag = BLT
					BLT = {
						NOT = { has_idea = BLT_temple_guard_idea }
						has_idea_with_trait = BLT_watcher_of_the_gate
						BLT_boltrus_is_gallant = yes
						BLT_boltrus_is_persuasive = yes
					}
				}
				BLT = { add_ideas = BLT_temple_guard_idea }
			}

			#Zohra's advisor ability
			if = {
				limit = {
					original_tag = BLT
					has_idea_with_trait = BLT_everstrider_cultist
				}
				every_character = {
					limit = {
						is_unit_leader = yes
						NOT = { has_trait = chosen_of_arakash }
						OR = {
							has_trait = camel_race_trait
							has_trait = cama_race_trait
						}
					}
					add_timed_unit_leader_trait = {
						trait = chosen_of_arakash
						days = 180
					}
				}
			}

			#Shahr's advisor ability
			if = {
				limit = {
					FROM = {
						original_tag = BLT
						has_idea_with_trait = BLT_scourge_of_the_saltpans
					}
				}
				BLT = {
					every_owned_state = {
						limit = {
							impassable = no
							OR = {
								has_state_category = rural
								has_state_category = pastoral
								has_state_category = wasteland
							}
						}
						add_dynamic_modifier = { modifier = BLT_scourge_assault_modifier }
					}
				}
			}
		}
	}

	on_ace_promoted = {
		effect = {
			#Toravash's advisor ability
			if = {
				limit = {
					ROOT = {
						original_tag = BLT
						has_idea_with_trait = BLT_air_enthusiast
					}
				}
				ROOT = {
					add_to_variable = { BLT_air_agility_factor = 0.05 }
				}
			}
		}
	}

	on_war = {
		effect = {
			if = {
				limit = {
					THIS = { has_country_flag = BLT_already_demobilized_economy }
				}
				THIS = { clr_country_flag = BLT_already_demobilized_economy }
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					ROOT = {
						original_tag = BLT
						has_idea_with_trait = BLT_storm_doctrine_theorist
					}
				}
				set_temp_variable = { boltrian_air_force_power_value = ROOT.deployed_airforce_manpower_k }
				set_temp_variable = { enemy_air_force_power_value = FROM.deployed_airforce_manpower_k }
				every_other_country = {
					limit = {
						OR = {
							is_in_faction_with = ROOT
							is_in_faction_with = FROM
							has_guaranteed = FROM
						}
					}
					if = {
						limit = { is_in_faction_with = ROOT }
						add_to_temp_variable = { boltrian_air_force_power_value = ROOT.deployed_airforce_manpower_k }
					}
					else = {
						add_to_temp_variable = { enemy_air_force_power_value = FROM.deployed_airforce_manpower_k }
					}
				}

				if = {
					limit = {
						check_variable = {
							var = boltrian_air_force_power_value
							value = enemy_air_force_power_value
							compare = greater_than_or_equals
						}
					}
					if = {
						limit = {
							BLT_boltrus_is_tactical = yes
							NOT = {
								any_collection_element = {
									collection = { input = collection:countries_hurt_by_storm_king }
									tag = FROM
								}
							}
						}
						every_other_country = {
							limit = {
								OR = {
									original_tag = FROM
									has_guaranteed = FROM
									is_in_faction_with = FROM
								}
							}
							add_ideas = BLT_inexperience_against_airships_idea
						}
					}

					if = {
						limit = {
							BLT_boltrus_is_ruthless = yes
							any_collection_element = {
								collection = { input = collection:countries_hurt_by_storm_king }
								tag = FROM
							}
						}
						every_other_country = {
							limit = {
								OR = {
									original_tag = FROM
									has_guaranteed = FROM
									is_in_faction_with = FROM
								}
							}
							add_ideas = BLT_stratospheric_shell_shock_idea
						}
					}
					else = {
						every_other_country = {
							limit = {
								OR = {
									original_tag = FROM
									has_guaranteed = FROM
									is_in_faction_with = FROM
								}
							}
							add_ideas = BLT_shock_and_awe_idea
						}
					}
				}
				else = { add_ideas = BLT_airship_defensive_protocols }
			}
			else_if = {
				limit = {
					FROM = {
						original_tag = BLT
						has_idea_with_trait = BLT_storm_doctrine_theorist
						BLT_boltrus_is_patient = yes
					}
				}
				FROM = { add_ideas = BLT_airship_defensive_protocols }
			}
		}
	}

	on_peace = {
		effect = {
			#Kavnorok's advisor ability
			if = {
				limit = { has_variable = var:BLT_skystriker_current_arty_bonus_value }
				subtract_from_variable = { BLT_army_artillery_attack_factor = var:BLT_skystriker_current_arty_bonus_value }
				set_variable = { BLT_skystriker_current_arty_bonus_value = 0 }
			}

			#Saz's advisor ability
			if = {
				limit = {
					original_tag = BLT
					has_idea_with_trait = BLT_civic_administrator
				}
				set_temp_variable = { funds_multiplier_value = 1 }
				if = {
					limit = { BLT_boltrus_is_gallant = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_erudite = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_tactical = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_conniving = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_diligent = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_austere = yes }
					add_to_temp_variable = { funds_multiplier_value = 1 }
				}
				set_temp_variable = {
					fund_points = {
						value = 150
						multiply = var:funds_multiplier_value
					}
				}
				random_military_industrial_organization = { add_mio_funds = var:fund_points }
			}

			#Masarai's advisor ability
			if = {
				limit = {
					original_tag = BLT
					has_idea_with_trait = BLT_mechanical_engineer
				}
				if = {
					limit = { has_idea = BLT_duneriders_idea_1 }
					mio:BLT_mobile_tank_company = { add_mio_funds = 600 }
				}
				else_if = {
					limit = { has_idea = BLT_duneriders_idea_2 }
					mio:BLT_mobile_tank_company = { add_mio_funds = 450 }
				}
				else_if = {
					limit = { has_idea = BLT_duneriders_idea_3 }
					mio:BLT_mobile_tank_company = { add_mio_funds = 300 }
				}
				else_if = {
					limit = { has_idea = BLT_duneriders_idea_4 }
					mio:BLT_mobile_tank_company = { add_mio_funds = 150 }
				}
			}
		}
	}

	on_army_leader_won_combat = {
		effect = {
			#Ryiz's advisor ability
			if = {
				limit = {
					FROM = {
						original_tag = BLT
						has_idea_with_trait = BLT_expert_skirmisher
						NOT = { has_idea = BLT_raiding_frenzy }
					}
					ROOT = { is_character = BLT_ryiz_stormrider }
				}
				FROM = {
					add_timed_idea = {
						idea = BLT_raiding_frenzy
						days = 25
					}
				}
			}
		}
	}

	on_unit_leader_level_up = {
		effect = {
			#Bevak's advisor ability
			if = {
				limit = {
					FROM = { has_idea_with_trait = BLT_seasoned_regiment_driller }
					ROOT = { is_army_leader = yes }
				}
				if = {
					limit = { BLT_boltrus_is_tactical = yes }
					set_temp_variable = { boltrus_is_tactical_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_conniving = yes }
					set_temp_variable = { boltrus_is_conniving_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_diligent = yes }
					set_temp_variable = { boltrus_is_diligent_value = 1 }
				}
				if = {
					limit = { BLT_boltrus_is_ruthless = yes }
					set_temp_variable = { boltrus_is_ruthless_value = 1 }
				}
				set_temp_variable = {
					extra_stat_chance = {
						value = 25
						add = {
							value = boltrus_is_tactical_value
							add = boltrus_is_conniving_value
							add = boltrus_is_diligent_value
							add = boltrus_is_ruthless_value
							multiply = 10
						}
					}
				}
				random = {
					chance = var:extra_stat_chance
					random_list = {
						1 = { add_attack = 1 }
						1 = { add_defense = 1 }
						1 = { add_logistics = 1 }
						1 = { add_planning = 1 }
					}
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			# Otabek's advisor ability
			if = {
				limit = {
					FROM = {
						original_tag = BLT
						has_idea_with_trait = BLT_temple_guard_banneret
					}
					ROOT = {
						NOT = { has_idea = BLT_heat_delirium }
					}
					if = {
						limit = {
							OR = {
								BLT_boltrus_is_brave = no
								BLT_boltrus_is_tactical = no
								BLT_boltrus_is_hardened = no
								BLT_boltrus_is_conniving = no
							}
						}
						FROM.FROM = { is_core_of = BLT }
					}
				}
				ROOT = { country_event = boltria.27 }
			}
		}
	}

	on_yearly_BLT = {
		effect = {
			#Boltrian Royal Development Charter concern ability
			if = {
				limit = {
					has_idea_with_trait = BLT_infrastructure_development
					any_owned_state = { infrastructure < 5 }
				}
				if = {
					limit = {
						BLT_boltrus_is_diligent = yes
						BLT_boltrus_is_persuasive = yes
					}
					set_temp_variable = { num_of_states_to_develop = 4 }
				}
				else = {
					set_temp_variable = { num_of_states_to_develop = 3}
				}
				set_temp_variable = { lvl_of_infrastructure_to_check = 0 }
				while_loop_effect = {
					limit = {
						check_variable = { num_of_states_to_develop > 0 }
						check_variable = {
							var = lvl_of_infrastructure_to_check
							value = 5
							compare = less_than_or_equals
						}
					}
					for_each_loop = {
						array = owned_states
						value = v
						break = break
						if = {
							limit = {
								check_variable = { num_of_states_to_develop = 0 }
							}
							set_temp_variable = { break = 1 }
						}
						if = {
							limit = {
								check_variable = { break = 0 }
								var:v = {
									is_controlled_by = BLT
									#infrastructure = var:lvl_of_infrastructure_to_check
									check_variable = { building_level@infrastructure = PREV.var:lvl_of_infrastructure_to_check }
								}
							}
							add_to_temp_variable = { num_of_states_to_develop = -1 }
							var:v = {
								add_building_construction = {
									level = 1
									type = infrastructure
									instant_build = yes
								}
							}
						}
					}
					add_to_temp_variable = { lvl_of_infrastructure_to_check = 1 }
				}
			}

			#Ashura's advisor ability
			if = {
				limit = { has_idea_with_trait = BLT_head_of_mathematics }
				if = {
					limit = { BLT_boltrus_is_erudite = yes }
					random_list = {
						1 = {
							add_tech_bonus = {
								name = BLT_head_of_mathematics
								bonus = 0.40
								uses = 1
								category = industry
							}
						}
						1 = {
							add_tech_bonus = {
								name = BLT_head_of_mathematics
								bonus = 0.40
								uses = 1
								category = electronics
							}
						}
					}
				}
				else = {
					random_list = {
						1 = {
							add_tech_bonus = {
								name = BLT_head_of_mathematics
								bonus = 0.25
								uses = 1
								category = industry
							}
						}
						1 = {
							add_tech_bonus = {
								name = BLT_head_of_mathematics
								bonus = 0.25
								uses = 1
								category = electronics
							}
						}
					}
				}
			}
		}
	}

	on_monthly_BLT = {
		effect = {
			#Erudite trait effect
			if = {
				limit = { BLT_boltrus_is_erudite = yes }
				every_military_industrial_organization = { add_mio_funds = num_of_civilian_factories }
			}
		}
	}

	on_weekly_BLT = {
		effect = {
			#Kavnorok's advisor ability
			if = {
				limit = {
					has_idea_with_trait = BLT_fire_saturation
					has_war = yes
					check_variable = {
						var = BLT_skystriker_current_arty_bonus_value
						value = @BLT_SKYSTRIKER_BONUS_CAP
						compare = less_than_or_equals
					}
				}
				add_to_variable = { BLT_army_artillery_attack_factor = 1 }
				add_to_variable = { BLT_skystriker_current_arty_bonus_value = 1 }
			}
		}
	}
}

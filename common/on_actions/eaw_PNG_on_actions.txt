on_actions = {
	on_startup = {
		effect = {
			PNG = {
				country_event = { id = penguin.0 }
			}
		}
	}
	#FROM is war target
	on_declare_war = {
		effect = {
			log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] on_declare_war"
			if = {
				limit = {
					original_tag = GRF
					FROM = {
						original_tag = GFF
					}
					PNG = {
						has_government = fascism
					}
				}
				PNG = {
					country_event = { id = penguin.41 hours = 6 }
				}
			}
		}
	}

	on_daily_PNG = {
		effect = {
			if = {
				limit = {
					has_country_flag = PNG_limited_war_flag
				}
				if = {
					limit = {
						288 = {
							owner = {
								has_war_with = PNG
								surrender_progress > 0.14
							}
						}
					}
					country_event = penguin.61
				}
				else_if = {
					limit = {
						PNG = {
							surrender_progress > 0.14
						}
						288 = {
							owner = {
								has_war_with = PNG
							}
						}
					}
					country_event = penguin.63
				}
			}
		}
	}
}

	on_weekly_LAK = {
		effect = {
			if = {
				limit = {
					has_variable = LAK_integration_progress_var
					has_variable = LAK_integration_weekly_progress
				}
				add_to_variable = { LAK_integration_progress_var = LAK_integration_weekly_progress }
				LAK_update_integration = yes
			}
			if = {
				limit = {
					has_country_leader = {
						character = LAK_arwyn_alister
						ruling_only = yes
					}
				}
				clear_array = LAK_BRD_enemies_array
				if = {
					limit = {
						has_completed_focus = LAK_BRD_proclaim_our_intentions
					}
					every_other_country = {
						limit = {
							LAK_BRD_current_and_potential_enemies = yes
						}
						add_to_array = { ROOT.LAK_BRD_enemies_array = THIS }
					}
				}
				find_highest_in_array = {
					array = LAK_BRD_modifiers_array
					index = temp_index
				}
				set_variable = {
					LAK_BRD_modifiers_array^temp_index = {
						value = LAK_BRD_modifiers_array^temp_index
						subtract = {
							value = LAK_BRD_overcap
							divide = 100
						}
					}
				}
				every_core_state = {
					limit = {
						is_owned_and_controlled_by_ROOT_or_subject = yes
					}
					add_to_temp_variable = {
						temp_population = {
							value = state_population_k
							divide = 1000
						}
					}
				}
				set_variable = {
					LAK_BRD_modifiers_gain_speed = {
						value = 0.01
						add = {
							value = temp_population
							subtract = 0.5
							round = yes
							multiply = 0.005
						}
					}
				}
				for_loop_effect = {
					end = LAK_BRD_modifiers_temp_array^num
					if = {
						limit = {
							check_variable = { LAK_BRD_modifiers_temp_array^v > LAK_BRD_modifiers_array^v }
						}
						add_to_variable = { LAK_BRD_modifiers_array^v = LAK_BRD_modifiers_gain_speed }
					}
				}
				LAK_BRD_recalculate_modifiers = yes
				count_inflicted_casualties_on_current_enemies = yes
			}
		}
	}

							is_neighbor_of_ROOT_or_ally = yes

						random_unit_leader = {
							limit = {

								NOT = { has_trait = wounded }
							}
							add_timed_unit_leader_trait = {
								trait = wounded

							}
						}
					}
				}
			}
		}

on_actions = {
	on_startup = {
		effect = {
			384 = {
				force_enable_resistance = BOI
				start_resistance = BOI
				add_compliance = 50
				add_resistance = 0
				add_dynamic_modifier = { modifier = LAK_BRD_activity_dm }
			}
			679 = {
				force_enable_resistance = BOI
				start_resistance = BOI
				add_compliance = 50
				add_resistance = 0
				add_dynamic_modifier = { modifier = LAK_BRD_activity_dm }
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = LAK
						is_subject = no
						has_country_flag = LAK_reclamation_war_flag
					}
					FROM.FROM = {
						OR = {
							state = 384
							state = 679
						}
					}
					controls_state = 384
					controls_state = 679
				}
				ROOT = { 
					if = {
						limit = {
							has_country_flag = LAK_selected_empire_path_flag
						}
						country_event = { id = rumare_first_act.67 days = 1 }
						set_country_flag = LAK_empire_path_flag
						clr_country_flag = LAK_selected_empire_path_flag
					}
					else = {
						country_event = { id = rumare_first_act.69 days = 1 }
					}
				}
			}
		}
	}
	on_weekly_LAK = {
		effect = {
			if = {
				limit = {
					has_variable = LAK_integration_progress_var
					has_variable = LAK_integration_weekly_progress
				}
				add_to_variable = { LAK_integration_progress_var = LAK_integration_weekly_progress }
				LAK_update_integration = yes
			}
			if = {
				limit = {
					has_country_leader = {
						character = LAK_arwyn_alister
						ruling_only = yes
					}
				}
				clear_array = LAK_BRD_enemies_array
				if = {
					limit = {
						has_completed_focus = LAK_BRD_proclaim_our_intentions
					}
					every_other_country = {
						limit = {
							LAK_BRD_current_and_potential_enemies = yes
						}
						add_to_array = { ROOT.LAK_BRD_enemies_array = THIS }
					}
					LAK_calculate_BRD_modifiers = yes
				}
			}
		}
	}

	on_monthly_LAK = {
		effect = {
			if = {
				limit = {
					has_completed_focus = LAK_BRD_doctrine_focus
				}
				LAK_BRD_doctrine_effect = yes
			}
		}
	}
	on_war = {
		effect = {
			if = { # If whoever declares war on BOI, you need to hurry up to kill Griffonstone
				limit = {
					original_tag = BOI
					LAK = {
						is_subject = no
						NOT = {
							has_war_with = BOI
							has_completed_focus = LAK_the_spark
							focus_progress = { focus = LAK_the_spark progress > 0.0 }
							owns_state = 384
							owns_state = 679
						}
					}
				}
				LAK = { country_event = rumare_first_act.23 }
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					tag = LAK
					has_country_flag = LAK_reclamation_war_flag
					has_country_flag = LAK_empire_path_flag
				}
				if = {
					limit = {
						384 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
						679 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
						389 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
					}
					country_event = rumare_empire.1
					clr_country_flag = LAK_reclamation_war_flag
				}
				else = {
					# some failsafe
				}
			}
		}
	}
	on_capitulation = { # TODO - make a proper thing to white peace WNG
		effect = {
			if = {
				limit = {
					FROM = {
						tag = LAK
						has_country_flag = LAK_reclamation_war_flag
					}
					ROOT = {
						tag = BOI
						has_war_together_with = WNG
					}
				}
				FROM = {
					set_country_flag = { flag = bypass_on_peaceconference_ended days = 1 value = 1 }
					every_enemy_country = { white_peace = PREV }
				}
			}
		}
	}
}

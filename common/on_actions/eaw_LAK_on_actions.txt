on_actions = {
	on_startup = {
		effect = {
			# Probably transfer these to state's files before release
			384 = {
				force_enable_resistance = BOI
				add_resistance = 30
				add_resistance_target = 50
				add_dynamic_modifier = { modifier = LAK_BRD_activity_dm }
			}
			679 = {
				force_enable_resistance = BOI
				add_resistance = 30
				add_resistance_target = 50
				add_dynamic_modifier = { modifier = LAK_BRD_activity_dm }
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = { # gives an appropriate event upon reclaiming mainland (Historical path OR Empire path)
				limit = {
					ROOT = {
						tag = LAK
						is_subject = no
						has_country_flag = LAK_reclamation_war_flag
					}
					FROM.FROM = {
						OR = {
							state = 384
							state = 679
						}
					}
					controls_state = 384
					controls_state = 679
				}
				ROOT = { 
					if = {
						limit = {
							has_country_flag = LAK_selected_empire_path_flag
						}
						country_event = { id = rumare_first_act.67 days = 1 }
						set_country_flag = LAK_empire_path_flag
						clr_country_flag = LAK_selected_empire_path_flag
					}
					else = {
						country_event = { id = rumare_first_act.69 days = 1 }
					}
				}
			}
			################
			### BRD PATH ###
			################
			if = {
				limit = {
					ROOT = {
						tag = LAK
						is_subject = no
						has_country_leader = {
							character = LAK_arwyn_alister
							ruling_only = yes
						}
					}
					meta_trigger = {
						text = {
							is_in_array = { ROOT.LAK_BRD_target_cities = [STATE] }
						}
						STATE = "[FROM.FROM.GetId]"
					}
				}
				FROM.FROM = {
					state_event = {
						id = rumare_BRD.12
						hours = 6
						trigger_for = ROOT
					}
					meta_effect = {
						text = {
							remove_from_array = { ROOT.LAK_BRD_target_cities = [STATE] }
						}
						STATE = "[THIS.GetId]"
					}
				}
			}
		}
	}
	on_weekly_LAK = {
		effect = {
			#######################
			### HISTORICAL PATH ###
			#######################
			if = {
				limit = {
					has_focus_tree = LAK_main_tree
				}
				if = { # Factions
					limit = {
						has_country_flag = LAK_factions_system_available
					}
					clamp_variable = { var = LAK_moderate_var min = 0 max = 100 }
					clamp_variable = { var = LAK_radical_var min = 0 max = 100 }
					set_variable = {
						LAK_moderate_tier = {
							value = 1
							if = {
								limit = {
									value = LAK_moderate_var greater_than_or_equals = 75
								}
								add = 2
							}
							else_if = {
								limit = {
									value = LAK_moderate_var greater_than_or_equals = 24
								}
								add = 1
							}
						}
						LAK_radical_tier = {
							value = 1
							if = {
								limit = {
									value = LAK_radical_var greater_than_or_equals = 75
								}
								add = 2
							}
							else_if = {
								limit = {
									value = LAK_radical_var greater_than_or_equals = 24
								}
								add = 1
							}
						}
					}
				}
				if = { # Integration
					limit = {
						has_variable = LAK_integration_progress_var
						has_variable = LAK_integration_weekly_progress
					}
					add_to_variable = { LAK_integration_progress_var = LAK_integration_weekly_progress }
					LAK_check_integration = yes
				}
				if = { # BRD sub-tree
					limit = {
						has_completed_focus = LAK_meet_arwyn
						NOT = {
							has_completed_focus = LAK_conspiracy_in_shadows
							has_country_flag = LAK_BRD_focus_activity_cd
							has_shine_effect_on_focus = LAK_conspiracy_in_shadows
							has_shine_effect_on_focus = LAK_terrorists_in_government
							has_shine_effect_on_focus = LAK_partisans_in_army
							has_shine_effect_on_focus = LAK_learn_from_BRD
						}
					}
					if = {
						limit = {
							NOT = {
								has_completed_focus = LAK_conspiracy_in_shadows
								has_shine_effect_on_focus = LAK_conspiracy_in_shadows
							}
							has_completed_focus = LAK_terrorists_in_government
							has_completed_focus = LAK_partisans_in_army
						}
						set_variable = { LAK_BRD_focus_selection_var = 4 }
					}
					else_if = {
						limit = {
							NOT = {
								has_completed_focus = LAK_terrorists_in_government
								has_shine_effect_on_focus = LAK_terrorists_in_government
							}
							has_completed_focus = LAK_learn_from_BRD
							has_completed_focus = LAK_partisans_in_army
						}
						set_variable = { LAK_BRD_focus_selection_var = 3 }
					}
					else_if = {
						limit = {
							NOT = {
								has_completed_focus = LAK_partisans_in_army
								has_shine_effect_on_focus = LAK_partisans_in_army
							}
							has_completed_focus = LAK_learn_from_BRD
						}
						set_variable = { LAK_BRD_focus_selection_var = 2 }
					}
					else_if = {
						limit = {
							NOT = {
								has_completed_focus = LAK_learn_from_BRD
								has_shine_effect_on_focus = LAK_learn_from_BRD
							}
						}
						set_variable = { LAK_BRD_focus_selection_var = 1 }
					}
					country_event = rumare_second_act.14
					LAK_BRD_focus_activity_cd_effect = yes
				}
			}
			###################
			### EMPIRE PATH ###
			###################
			if = {
				limit = {
					has_country_flag = LAK_empire_path_flag
				}
			}
			################
			### BRD PATH ###
			################
			if = {
				limit = {
					has_country_leader = {
						character = LAK_arwyn_alister
						ruling_only = yes
					}
				}
				clear_array = LAK_BRD_enemies_array
				if = {
					limit = {
						has_completed_focus = LAK_BRD_proclaim_our_intentions
					}
					every_other_country = {
						limit = {
							LAK_BRD_current_and_potential_enemies = yes
						}
						add_to_array = { ROOT.LAK_BRD_enemies_array = THIS }
					}
				}
				find_highest_in_array = {
					array = LAK_BRD_modifiers_array
					index = temp_index
				}
				set_variable = {
					LAK_BRD_modifiers_array^temp_index = {
						value = LAK_BRD_modifiers_array^temp_index
						subtract = {
							value = LAK_BRD_overcap
							divide = 100
						}
					}
				}
				every_core_state = {
					limit = {
						is_owned_and_controlled_by_ROOT_or_subject = yes
					}
					add_to_temp_variable = {
						temp_population = {
							value = state_population_k
							divide = 1000
						}
					}
				}
				set_variable = {
					LAK_BRD_modifiers_gain_speed = {
						value = 0.01
						add = {
							value = temp_population
							subtract = 0.5
							round = yes
							multiply = 0.005
						}
					}
				}
				for_loop_effect = {
					end = LAK_BRD_modifiers_temp_array^num
					if = {
						limit = {
							check_variable = { LAK_BRD_modifiers_temp_array^v > LAK_BRD_modifiers_array^v }
						}
						add_to_variable = { LAK_BRD_modifiers_array^v = LAK_BRD_modifiers_gain_speed }
						clamp_variable = {
							var = LAK_BRD_modifiers_array^v
							max = LAK_BRD_modifiers_temp_array^v
						}
					}
				}
				LAK_BRD_recalculate_modifiers = yes
				count_inflicted_casualties_on_current_enemies = yes
			}
		}
	}

	on_monthly_LAK = {
		effect = {
			#######################
			### HISTORICAL PATH ###
			#######################
			if = { # if you happen to have a defensive war against anywho, they'll white-peace if suffer too many casualties
				limit = {
					has_defensive_war = yes
					is_subject = no
					is_in_faction = no
					383 = { is_owned_and_controlled_by = ROOT } # island
					384 = { is_owned_and_controlled_by = ROOT } # mainland
					has_focus_tree = LAK_main_tree
					NOT = { has_country_flag = LAK_empire_path_flag }
				}
				every_enemy_country = {
					limit = {
						has_offensive_war_with = ROOT
						casualties_k > 100
						is_subject = no
						OR = {
							is_faction_leader = yes
							is_in_faction = no
						}
						NOT = { is_in_array = { ROOT.LAK_enemies_in_long_war = THIS } }
					}
					ROOT = {
						activate_targeted_decision = {
							target = PREV
							decision = LAK_the_long_war_mission
						}
						country_event = rumare_misc.2
					}
					activate_targeted_decision = {
						target = LAK
						decision = LAK_the_long_war_mission
					}
					country_event = rumare_misc.3
					add_to_array = { ROOT.LAK_enemies_in_long_war = THIS }
				}
			}
			###################
			### EMPIRE PATH ###
			###################
			if = {
				limit = {
					has_country_flag = LAK_empire_path_flag
				}
			}
			################
			### BRD PATH ###
			################
			if = {
				limit = {
					has_country_leader = {
						character = LAK_arwyn_alister
						ruling_only = yes
					}
				}
				if = { # bonus for doctrine mastery if any of enemies have more mastery than you
					limit = {
						has_completed_focus = LAK_BRD_doctrine_focus
					}
					check_army_mastery_level = yes
					for_each_scope_loop = {
						array = LAK_BRD_enemies_array
						check_army_mastery_level = yes
						if = {
							limit = {
								any_of = {
									array = doctrine_mastery_array
									check_variable = { v > ROOT.doctrine_mastery_array^i }
								}
							}
							for_loop_effect = {
								start = 0
								end = 4
								if = {
									limit = {
										check_variable = { doctrine_mastery_array^v > ROOT.doctrine_mastery_array^v }
									}
									add_to_temp_variable = { temp_country_has_better_doctrine^v = 1 }
									log = "[THIS.GetName] has track [?v|Y0] better: [?doctrine_mastery_array^v] > [?ROOT.doctrine_mastery_array^v]"
								}
							}
						}
						clear_array = doctrine_mastery_array
					}
					if = {
						limit = {
							has_subdoctrine_in_track = infantry
							check_variable = { temp_country_has_better_doctrine^0 > 0 }
						}
						set_variable = {
							LAK_army_dm_infantry_track_mastery_gain_factor = {
								value = temp_country_has_better_doctrine^0
								multiply = 0.05
							}
						}
					}
					else = {
						clear_variable = LAK_army_dm_infantry_track_mastery_gain_factor
					}
					if = {
						limit = {
							has_subdoctrine_in_track = combat_support
							check_variable = { temp_country_has_better_doctrine^1 > 0 }
						}
						set_variable = {
							LAK_army_dm_combat_support_track_mastery_gain_factor = {
								value = temp_country_has_better_doctrine^1
								multiply = 0.05
							}
						}
					}
					else = {
						clear_variable = LAK_army_dm_combat_support_track_mastery_gain_factor
					}
					if = {
						limit = {
							has_subdoctrine_in_track = armor
							check_variable = { temp_country_has_better_doctrine^2 > 0 }
						}
						set_variable = {
							LAK_army_dm_armor_track_mastery_gain_factor = {
								value = temp_country_has_better_doctrine^2
								multiply = 0.05
							}
						}
					}
					else = {
						clear_variable = LAK_army_dm_armor_track_mastery_gain_factor
					}
					if = {
						limit = {
							has_subdoctrine_in_track = operations
							check_variable = { temp_country_has_better_doctrine^3 > 0 }
						}
						set_variable = {
							LAK_army_dm_operations_track_mastery_gain_factor = {
								value = temp_country_has_better_doctrine^3
								multiply = 0.05
							}
						}
					}
					else = {
						clear_variable = LAK_army_dm_operations_track_mastery_gain_factor
					}
					clear_array = doctrine_mastery_array
				}
				if = { # 25% monthly chance to would enemy's general
					limit = {
						has_war = yes
						has_completed_focus = LAK_BRD_work_outside
					}
					random = {
						chance = 25
						random_enemy_country = {
							limit = {
								is_neighbor_of_ROOT_or_ally = yes
							}
							random_unit_leader = {
								limit = {
									is_leading_army = yes
									NOT = { has_trait = wounded }
								}
								add_timed_unit_leader_trait = {
									trait = wounded
									days = 90
								}
							}
						}
					}
				}
			}
		}
	}
	on_war = {
		effect = {
			if = { # If whoever declares war on BOI, you need to hurry up to kill Griffonstone
				limit = {
					original_tag = BOI
					LAK = {
						is_subject = no
						NOT = {
							has_war_with = BOI
							has_completed_focus = LAK_the_spark
							focus_progress = { focus = LAK_the_spark progress > 0.0 }
							owns_state = 384
							owns_state = 679
						}
					}
				}
				LAK = { country_event = rumare_first_act.23 }
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					tag = LAK
					has_country_flag = LAK_reclamation_war_flag
					has_country_flag = LAK_empire_path_flag
				}
				if = {
					limit = {
						384 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
						679 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
						389 = { is_owned_and_controlled_by_ROOT_or_subject = yes }
					}
					country_event = rumare_empire.1
					clr_country_flag = LAK_reclamation_war_flag
				}
				else = {
					# some failsafe
				}
			}
		}
	}
	on_capitulation = { # TODO - make a proper thing to white peace WNG
		effect = {
			if = {
				limit = {
					FROM = {
						tag = LAK
						has_country_flag = LAK_reclamation_war_flag
					}
					ROOT = {
						tag = BOI
						has_war_together_with = WNG
					}
				}
				FROM = {
					set_country_flag = { flag = bypass_on_peaceconference_ended days = 1 value = 1 }
					every_enemy_country = { white_peace = PREV }
				}
			}
		}
	}
	on_puppet = {
		effect = {
			if = { # LAK abd BOI can't puppet each other
				limit = {
					ROOT = {
						OR = {
							original_tag = BOI
							original_tag = LAK
						}
					}
					FROM = {
						OR = {
							original_tag = BOI
							original_tag = LAK
						}
					}
				}
				ROOT = {
					annex_country = {
						target = FROM
						transfer_troops = yes
					}
				}
			}
		}
	}

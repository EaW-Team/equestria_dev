on_actions = {

	#on_capitulation_immediate = {
	on_before_peace_conference_start = {
		effect = {
			if = {
				limit = {
					FROM = { any_owned_state = { is_coastal = yes } }
					ROOT = {
						NOT = {
							any_owned_state = {
								is_coastal = yes
							}
						}
					}
				}
				ROOT = { set_country_flag = EAW_landlock_navy_target }
				set_global_flag = EAW_transfer_navy_to_landlock
				FROM = { transfer_navy = { target = ZZZ } }
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			# PUT EVERYTHING IN THE IF BELOW
			# THE bypass_on_peaceconference_ended FLAG **HAS** TO BE SET BEFORE CALLING THE white_peace COMMAND AND CLEARED AFTERWARDS
			if = {
				limit = {
					has_global_flag = EAW_transfer_navy_to_landlock
					ROOT = {
						has_country_flag = EAW_landlock_navy_target
					}
					NOT = { FROM = { is_subject_of = ROOT } }
				}
				ZZZ = { transfer_navy = { target = ROOT } }
				clr_global_flag = EAW_transfer_navy_to_landlock
				ROOT = { clr_country_flag = EAW_landlock_navy_target }
			}
			if = {
				limit = {
					has_global_flag = EAW_transfer_navy_to_landlock
					ROOT = {
						has_country_flag = EAW_landlock_navy_target
					}
					FROM = { is_subject_of = ROOT }
				}
				ZZZ = { transfer_navy = { target = FROM } }
				clr_global_flag = EAW_transfer_navy_to_landlock
				ROOT = { clr_country_flag = EAW_landlock_navy_target }
			}
		}
	}
}

on_actions = {
	on_before_peace_conference_start = {
		effect = {
			set_country_flag = actively_tracking_war_casualties
			set_variable = {
				current_war_casualties_k = casualties_k
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					has_country_flag = actively_tracking_war_casualties
				}
				add_to_variable = {
					past_casualties_k = current_war_casualties_k
				}
				subtract_from_variable = {
					past_casualties_k = casualties_k
				}
				clear_variable = current_war_casualties_k
			}
		}
	}
}

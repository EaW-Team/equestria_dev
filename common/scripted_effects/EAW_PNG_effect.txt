PNG_empty_yak_claimed_states = {
	custom_effect_tooltip = PNG_empty_yak_claimed_states_tt
	every_owned_state = {
		limit = {
			is_fully_controlled_by = ROOT
			OR = {
				is_claimed_by = YAK
				is_claimed_by = JAK
			}
		}
		if = {
			limit = {
				industrial_complex > 0
			}
			set_temp_variable = {
				amount_civ_moved = industrial_complex_level
			}
			remove_building = {
				type = industrial_complex
				level = amount_civ_moved
			}
			ROOT = {
				random_core_state = {
					add_extra_state_shared_building_slots = amount_civ_moved
					add_building_construction = {
						type = industrial_complex
						level = amount_civ_moved
						instant_build = yes
					}
				}
			}
		}
		if = {
			limit = {
				arms_factory > 0
			}
			set_temp_variable = {
				amount_mil_moved = arms_factory_level
			}
			remove_building = {
				type = arms_factory
				level = amount_mil_moved
			}
			ROOT = {
				random_core_state = {
					add_extra_state_shared_building_slots = amount_mil_moved
					add_building_construction = {
						type = arms_factory
						level = amount_mil_moved
						instant_build = yes
					}
				}
			}
		}
	}
}

	custom_effect_tooltip = PNG_move_griffoid_industry_tt

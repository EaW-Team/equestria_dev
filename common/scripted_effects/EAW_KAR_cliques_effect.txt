
KAR_radar_graph_start = {
	# Set the initial positions with the original values
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_init = 0 }
	}

	clear_array = KAR_radar_graph_values_pct
	add_to_array = { KAR_radar_graph_values_pct = 1 }
	add_to_array = { KAR_radar_graph_values_pct = 1 }
	add_to_array = { KAR_radar_graph_values_pct = 1 }
	add_to_array = { KAR_radar_graph_values_pct = 1 }
	add_to_array = { KAR_radar_graph_values_pct = 1 }
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_final = KAR_radar_graph_values_pct^v }
	}
	
	calculate_radar_graph_5axis_anim_frame = yes

	clear_array = KAR_radar_graph_frame_array
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_array = { KAR_radar_graph_frame_array = sector_frame^v }
	}
}

# args (temp var): temp_axis_index, temp_change
KAR_add_radar_graph_value = {
	# Set the initial positions with the original values
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_init = KAR_radar_graph_values_pct^v }
	}

	# Set the new values
	add_to_variable = { KAR_radar_graph_values_pct^temp_axis_index = temp_change }
	clamp_variable = { var = KAR_radar_graph_values_pct^temp_axis_index min = 0 max = 1 }
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_temp_array = { axis_final = KAR_radar_graph_values_pct^v }
	}
	
	# Call the scripted effect
	calculate_radar_graph_5axis_anim_frame = yes

	# Set the frame data to the output of the scripted effect
	clear_array = KAR_radar_graph_frame_array
	for_loop_effect = {
		start = 0 end = 5 value = v
		add_to_array = { KAR_radar_graph_frame_array = sector_frame^v }
	}
}

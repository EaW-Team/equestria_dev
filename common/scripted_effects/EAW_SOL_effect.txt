# INNER CIRCLE SCRIPTED EFFECTS #
# For the sake of saving space in the events file, each member of the inner circle gets a scripted effect like this
# The timers are set in the events file
# Generally should be understanable without more comments, as this checks for if a specific focus not being completed, and creates a shine on it if it's not
@SOL_inner_circle_flag_timer = 71
@SOL_inner_circle_timer = 70
SOL_inner_circle_aunt_orange_shines = {
    if = {
        limit = {
            NOT = { has_completed_focus = SOL_laying_seed }
        }
        activate_shine_on_focus = SOL_laying_seed
        set_country_flag = {
            flag = SOL_laying_seed_in_progress_flag
            days = @SOL_inner_circle_flag_timer
            value = 1
        }
        country_event = {
            id = solarempire_inner_circle_orange.2
            days = @SOL_inner_circle_timer
        }

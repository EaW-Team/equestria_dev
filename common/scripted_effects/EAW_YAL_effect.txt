			remove_idea = YAL_economic_downturn
			add_idea = YAL_sluggish_economy
		}
	}
	else_if = {
		limit = {
			has_idea = YAL_sluggish_economy
		}
		swap_ideas = {
			remove_idea = YAL_sluggish_economy
			add_idea = YAL_sluggish_economy_2
		}
	}
	else_if = {
		limit = {
			has_idea = YAL_sluggish_economy_2
		}
		swap_ideas = {
			remove_idea = YAL_sluggish_economy_2
			add_idea = YAL_lingering_economic_woes
		}
	}
	else_if = {
		limit = {
			has_idea = YAL_lingering_economic_woes
		}
		remove_ideas = YAL_lingering_economic_woes
	}
}

YAL_karl_modify_influence_and_loyalty = {
	# Calculate the influence changes
	
	# syndie_effective_gain = syndie_gain - syndie_influence * (syndie_gain + anarch_gain + moderate_gain)
	# Example: 
	# if syndie_gain is 0.3 (30% gain) and all other gains are 0
	# syndie_influence is 0.3 (30%), anarch_influence is 0.3, and moderate influece is 0.4
	# then syndie_effective_gain is 0.3 - 0.3 * 0.3 = 0.21
	# Meanwhile anarch_effective_gain is - 0.3 * 0.3 = -0.09
	# And moderate_effective_gain is -0.4 * 0.3 = -0.12
	# Which all adds up to 0 exactly.

	# However, moderate influence is capped at 0.4
	# So we need to calculate max moderate influence gain, which can be done by calculuating the max influence
	# moderate_max_gain = (0.4 - moderate_influence * (1 - syndie_gain - anarch_gain))/(1 - moderate_influence)

	# Example: syndie_gain = 0.1, anarch_gain = 0.1, moderate_gain = 0.2
	# syndie_influence = 0.35, anarch_influence = 0.4, moderate_influence = 0.35
	# Then: moderate_max_gain = 0.1846
	# And from that moderate_effective_gain = 0.05, just how it should be.

	clamp_temp_variable = {
		var = syndie_gain
		max = 1
		min = 0
	}
	clamp_temp_variable = {
		var = anarch_gain
		max = 1
		min = 0
	}

	# if moderate_gain > moderate_max gain, we need to cap it at moderate_max_gain
	set_temp_variable = {
		moderate_max_gain = {
			value = 0.4
			subtract = {
				value = moderate_influence
				multiply = {
					value = 1
					subtract = syndie_gain
					subtract = anarch_gain
				}
			}
			divide = {
				value = 1
				subtract = moderate_influence
			}
		}
	}

	log = "max mod gain [?moderate_max_gain]"
	
	clamp_temp_variable = {
		var = moderate_gain
		min = 0
		max = moderate_max_gain
	}

	set_temp_variable = {
		moderate_effective_gain = {
			value = moderate_gain
			subtract = {
				value = moderate_influence
				multiply = {
					value = moderate_gain
					add = syndie_gain
					add = anarch_gain
				}
			}
		}
	}

	set_temp_variable = {
		syndie_effective_gain = {
			value = syndie_gain
			subtract = {
				value = syndie_influence
				multiply = {
					value = moderate_gain
					add = syndie_gain
					add = anarch_gain
				}
			}
		}
	}

	set_temp_variable = {
		anarch_effective_gain = {
			value = anarch_gain
			subtract = {
				value = syndie_influence
				multiply = {
					value = moderate_gain
					add = syndie_gain
					add = anarch_gain
				}
			}
		}
	}

	log = "gains: [?moderate_effective_gain]  [?syndie_effective_gain]  [?anarch_effective_gain]"

	add_to_variable = {
		moderate_influence = moderate_effective_gain
	}
	add_to_variable = {
		syndie_influence = syndie_effective_gain
	}
	add_to_variable = {
		anarch_influence = anarch_effective_gain
	}

	set_temp_variable = {
		moderate_loyalty_effective_gain = {
			value = moderate_loyalty_gain
			add = {
				value = moderate_effective_gain
				multiply = 0.5
			}
		}
	}

	set_temp_variable = {
		syndie_loyalty_effective_gain = {
			value = syndie_loyalty_gain
			add = {
				value = syndie_effective_gain
				multiply = 0.5
			}
		}
	}

	set_temp_variable = {
		anarch_loyalty_effective_gain = {
			value = anarch_loyalty_gain
			add = {
				value = anarch_effective_gain
				multiply = 0.5
			}
		}
	}

	add_to_variable = {
		moderate_loyalty = moderate_loyalty_effective_gain
	}
	add_to_variable = {
		syndie_loyalty = syndie_loyalty_effective_gain
	}
	add_to_variable = {
		anarch_loyalty = anarch_loyalty_effective_gain
	}

	log = "loy gains: [?moderate_loyalty_effective_gain]  [?syndie_loyalty_effective_gain]  [?anarch_loyalty_effective_gain]"

	custom_effect_tooltip = YAL_faction_change_tt
}

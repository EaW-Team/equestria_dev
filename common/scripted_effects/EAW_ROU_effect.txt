ROU_activate_di_karthin_principles = {
	add_dynamic_modifier = {
		modifier = ROU_principles_dm
	}
	set_country_flag = ROU_di_karthin_principles_active
	ROU_update_principles_GUI = yes
}
ROU_liberty_idealist = {
	custom_effect_tooltip = {
		localization_key = ROU_principle_idealist_effect_tt
		PRINCIPLE = ROU_principle_1_text
	}
	clear_variable = ROU_liberty_status_var
	ROU_update_principles_GUI = yes
}

		localization_key = ROU_principle_realist_effect_tt

	}
	add_to_variable = { ROU_democracy_status_var = 1 }
	ROU_update_principles_GUI = yes
}
ROU_update_principles_GUI = {
	if = {
		limit = {
			ROU_liberty_is_realist = yes
			ROU_prosperity_is_realist = yes
			ROU_democracy_is_realist = yes
		}
		# 3 realist // 0 idealist
		set_variable = { ROU_principles_dm_stability_factor = -0.1 }
		set_variable = { ROU_principles_dm_war_support_factor = 0.3 }
		set_variable = { ROU_principles_dm_industrial_capacity_factory = 0.3 }
	}
	else_if = {
		limit = {
			count_triggers = {
				amount = 2
				ROU_liberty_is_realist = yes
				ROU_prosperity_is_realist = yes
				ROU_democracy_is_realist = yes
			}
		}
		# 2 realist // 1 idealist
		set_variable = { ROU_principles_dm_stability_factor = -0.05 }
		set_variable = { ROU_principles_dm_war_support_factor = 0.15 }
		set_variable = { ROU_principles_dm_industrial_capacity_factory = 0.15 }
	}
	else_if = {
		limit = {
			OR = {
				ROU_liberty_is_realist = yes
				ROU_prosperity_is_realist = yes
				ROU_democracy_is_realist = yes
			}
		}
		# 1 realist // 2 idealist
		set_variable = { ROU_principles_dm_stability_factor = 0.05 }
		set_variable = { ROU_principles_dm_war_support_factor = 0.05 }
		set_variable = { ROU_principles_dm_industrial_capacity_factory = 0.05 }
	}
	else = {
		# 0 realist // 3 idealist
		set_variable = { ROU_principles_dm_stability_factor = 0.1 }
		set_variable = { ROU_principles_dm_war_support_factor = -0.05 }
		set_variable = { ROU_principles_dm_industrial_capacity_factory = -0.05 }
	}
	add_to_variable = { ROU_gui_dirty = 1 }
}

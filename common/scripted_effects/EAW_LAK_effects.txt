					10 = { country_event = rumare_factions.5 modifier = { factor = 0 has_variable = rumare_event_fired^5 } modifier = { factor = 0 check_variable = { temp_threat < 3 } } }
				}
			}
			40 = { # Chance of Moderates doing their move
				random_list = {
					10 = { country_event = rumare_factions.6 modifier = { factor = 0 has_variable = rumare_event_fired^6 } }
					10 = { country_event = rumare_factions.7 modifier = { factor = 0 has_variable = rumare_event_fired^7 } }

LAK_update_GUI = {
	clamp_variable = { var = LAK_moderate_var min = 0 max = 100 }
	clamp_variable = { var = LAK_radical_var min = 0 max = 100 }
	set_variable = {
		LAK_moderate_tier = {
			value = 1
			if = {
				limit = {
					value = LAK_moderate_var greater_than_or_equals = 75
				}
				add = 2
			}
			else_if = {
				limit = {
					value = LAK_moderate_var greater_than_or_equals = 24
				}
				add = 1
			}
		}
		LAK_radical_tier = {
			value = 1
			if = {
				limit = {
					value = LAK_radical_var greater_than_or_equals = 75
				}
				add = 2
			}
			else_if = {
				limit = {
					value = LAK_radical_var greater_than_or_equals = 24
				}
				add = 1
			}
		}
	}
	add_to_variable = { LAK_GUI_dirty = 1 }
}
LAK_update_integration = {
	clamp_variable = { var = LAK_integration_progress_var min = 0 max = 100 }
	if = {
		limit = {
			check_variable = { LAK_integration_progress_var = 100 }
		}
		# TODO
		set_country_flag = LAK_integration_finished
		clr_country_flag = LAK_integration_GUI_visible_flag
		clear_variable = LAK_integration_progress_var
	}
	add_to_variable = { LAK_GUI_dirty = 1 }
}

LAK_calculate_BRD_modifiers = {
	clear_variable = temp_industry
	clear_variable = temp_army
	every_enemy_country = {
		add_to_variable = { ROOT.temp_industry = num_of_military_factories }
		add_to_variable = { ROOT.temp_army = deployed_army_manpower_k }
	}
	log = "Industry: [?temp_industry]"
	log = "Army: [?temp_army]k"
	set_variable = {
		industry_ratio = {
			value = temp_industry
			divide = num_of_military_factories
		}
	}
	set_variable = {
		LAK_BRD_economy_law_dm_production_factory_max_efficiency_factor = {
			value = industry_ratio
			divide = 50
			clamp = {
				min = 0
				max = 1
			}
		}
	}
	set_variable = {
		army_ratio = {
			value = temp_army
			divide = deployed_army_manpower_k
		}
	}
	set_variable = {
		LAK_BRD_conscription_law_dm_army_attack_factor = {
			value = army_ratio
			divide = 100
			clamp = {
				min = 0
				max = 1
			}
		}
	}
	set_variable = { LAK_BRD_conscription_law_dm_army_org_factor = LAK_BRD_conscription_law_dm_army_attack_factor }
	log = "Industry: [?industry_ratio] times more"
	log = "Army: [?army_ratio] times"
}

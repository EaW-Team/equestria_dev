LAK_update_GUI = {
	clamp_variable = { var = LAK_moderate_var min = 0 max = 100 }
	clamp_variable = { var = LAK_radical_var min = 0 max = 100 }
	set_variable = {
		LAK_moderate_tier = {
			value = 1
			if = {
				limit = {
					value = LAK_moderate_var greater_than_or_equals = 75
				}
				add = 2
			}
			else_if = {
				limit = {
					value = LAK_moderate_var greater_than_or_equals = 24
				}
				add = 1
			}
		}
		LAK_radical_tier = {
			value = 1
			if = {
				limit = {
					value = LAK_radical_var greater_than_or_equals = 75
				}
				add = 2
			}
			else_if = {
				limit = {
					value = LAK_radical_var greater_than_or_equals = 24
				}
				add = 1
			}
		}
	}
	add_to_variable = { LAK_GUI_dirty = 1 }
}

LAK_calculate_BRD_modifiers = {
	clear_variable = total_industry
	clear_variable = total_army
	clear_variable = hostile_industry
	clear_variable = hostile_army
	every_other_country = {
		limit = {
			LAK_BRD_current_and_potential_enemies = yes
		}
		add_to_variable = { ROOT.hostile_industry = num_of_military_factories }
		add_to_variable = { ROOT.hostile_army = deployed_army_manpower_k }
	}
	set_variable = {
		industry_ratio = {
			value = hostile_industry
			divide = num_of_military_factories
			subtract = 1
			divide = 100
		}
	}
	set_variable = {
		army_ratio = {
			value = hostile_army
			divide = deployed_army_manpower_k
			subtract = 1
			divide = 100
		}
	}
	set_variable = {
		LAK_BRD_economy_law_dm_industrial_capacity_factory = {
			value = industry_ratio
			clamp = {
				min = 0
				max = 1
			}
		}
	}
	set_variable = {
		LAK_BRD_conscription_law_dm_army_attack_factor = {
			value = army_ratio
			clamp = {
				min = 0
				max = 0.5
			}
		}
	}
	set_variable = { LAK_BRD_conscription_law_dm_army_defence_factor = LAK_BRD_conscription_law_dm_army_attack_factor }
}

LAK_BRD_doctrine_effect = {
	check_army_mastery_level = yes
	every_other_country = {
		limit = {
			LAK_BRD_current_and_potential_enemies = yes
		}
		check_army_mastery_level = yes
		if = {
			limit = {
				OR = {
					check_variable = { doctrine_mastery_array^1 > ROOT.doctrine_mastery_array^1 }
					check_variable = { doctrine_mastery_array^2 > ROOT.doctrine_mastery_array^2 }
					check_variable = { doctrine_mastery_array^3 > ROOT.doctrine_mastery_array^3 }
					check_variable = { doctrine_mastery_array^4 > ROOT.doctrine_mastery_array^4 }
				}
			}
			for_loop_effect = {
				start = 1
				end = 5
				if = {
					limit = {
						check_variable = { doctrine_mastery_array^v > ROOT.doctrine_mastery_array^v }
					}
					add_to_temp_variable = { temp_country_has_better_doctrine^v = 1 }
					log = "[THIS.GetName] has track [?v|Y0] better: [?doctrine_mastery_array^v] > [?ROOT.doctrine_mastery_array^v]"
				}
				clear_variable = doctrine_mastery_array^v
			}
		}
	}
	if = {
		limit = {
			has_subdoctrine_in_track = infantry
			check_variable = { temp_country_has_better_doctrine^1 > 0 }
		}
		set_variable = {
			LAK_army_dm_infantry_track_mastery_gain_factor = {
				value = temp_country_has_better_doctrine^1
				multiply = 0.05
			}
		}
	}
	else = {
		clear_variable = LAK_army_dm_infantry_track_mastery_gain_factor
	}
	if = {
		limit = {
			has_subdoctrine_in_track = combat_support
			check_variable = { temp_country_has_better_doctrine^2 > 0 }
		}
		set_variable = {
			LAK_army_dm_combat_support_track_mastery_gain_factor = {
				value = temp_country_has_better_doctrine^2
				multiply = 0.05
			}
		}
	}
	else = {
		clear_variable = LAK_army_dm_combat_support_track_mastery_gain_factor
	}
	if = {
		limit = {
			has_subdoctrine_in_track = armor
			check_variable = { temp_country_has_better_doctrine^3 > 0 }
		}
		set_variable = {
			LAK_army_dm_armor_track_mastery_gain_factor = {
				value = temp_country_has_better_doctrine^3
				multiply = 0.05
			}
		}
	}
	else = {
		clear_variable = LAK_army_dm_armor_track_mastery_gain_factor
	}
	if = {
		limit = {
			has_subdoctrine_in_track = operations
			check_variable = { temp_country_has_better_doctrine^4 > 0 }
		}
		set_variable = {
			LAK_army_dm_operations_track_mastery_gain_factor = {
				value = temp_country_has_better_doctrine^4
				multiply = 0.05
			}
		}
	}
	else = {
		clear_variable = LAK_army_dm_operations_track_mastery_gain_factor
	}
	clear_variable = doctrine_mastery_array^1
	clear_variable = doctrine_mastery_array^2
	clear_variable = doctrine_mastery_array^3
	clear_variable = doctrine_mastery_array^4
}

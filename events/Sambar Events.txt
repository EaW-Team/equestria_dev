	desc = sambar_news.1.d

	is_triggered_only = yes
	
	major = yes

	option = {	  
		name = sambar_news.1.a
		trigger = {
			original_tag = SAM
		}

		trigger = {
			OR = {
				has_government = neutrality

			}
			NOT = { original_tag = SAM }
		}
	}
	option = {	  
		name = sambar_news.1.c
		trigger = {
			OR = {
				has_government = communism
				has_government = democratic
			}
			NOT = { original_tag = SAM }
		}
	}

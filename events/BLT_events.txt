			}

			fail_chance = {
				value = 100
				subtract = var:success_chance
			}
		}

				hidden_effect = { country_event = boltria.sandstorm_rescue.3 }

			limit = { BLT_boltrus_is_tactical = yes }
			add_to_temp_variable = {
				success_chance = 30

			}
		}
		set_temp_variable = {
			fail_chance = {
				value = 100

			}
		}

				custom_effect_tooltip = boltria_roll_success_generic

			}
